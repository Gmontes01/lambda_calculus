// 1. Types
enum class NodeType { VARIABLE, LAMBDA, APPLICATION };

// 2. Struct Declaration (Signatures Only)
struct lambda_expression {
    NodeType type;
    std::string name;

    lambda_expression* expression_1;
    lambda_expression* expression_2;

    static lambda_expression* deepcopy(const lambda_expression* node);
    bool is_valid() const;
    bool is_redux();
    
    void rename(const std::string& old_name, const std::string& new_name);
    void rename_free(const std::string& old_name, const std::string& new_name);
    void rename_bound(const std::string& old_name, const std::string& new_name);
    
    bool occurs_free_in(const std::string& var) const;
    std::string generate_fresh_name(const std::string& base, const lambda_expression* replacement) const;
    void substitute(const std::string& var, const lambda_expression* replacement);
    
    void reduce_total(int MAX_DEPTH = 1000);


    bool reduce();
    std::string to_string() const;
    
    ~lambda_expression();

    bool operator==(const lambda_expression& other) const;//alpha equivalence up to combinator variable names
    bool beta_equivalence(const lambda_expression& other, int max_depth) const;//beta equivalence
};

#include <iostream>
#include <sstream>
#include <stdexcept>
#include <new> // Required for std::bad_alloc

using namespace std;

// ==========================================
// STRING PARSING HELPERS
// ==========================================

string trim(const string& s) {
    size_t start = s.find_first_not_of(" \t\r\n");
    if (start == string::npos) return "";
    size_t end = s.find_last_not_of(" \t\r\n");
    return s.substr(start, end - start + 1);
}

bool is_valid_var_name(const string& s) {
    if (s.empty()) return false;
    for (char c : s) {
        if (!isalpha(c) && c != '_') return false;
    }
    return true;
}

bool is_separator(const string& line) {
    string t = trim(line);
    if (t.length() < 3) return false;
    for (char c : t) {
        if (c != '*') return false;
    }
    return true;
}

string remove_comments(const string& source) {
    string result = "";
    bool in_comment = false;
    
    for (size_t i = 0; i < source.length(); ++i) {
        // Look ahead to see if the next three characters are quotes
        if (i + 2 < source.length() && source[i] == '"' && source[i+1] == '"' && source[i+2] == '"') {
            in_comment = !in_comment; // Toggle the state (Open -> Close, or Close -> Open)
            i += 2; // Skip the next two quotes so we don't process them again
        } 
        // Only keep characters if we are NOT inside a comment
        else if (!in_comment) {
            result += source[i]; 
        }
    }
    
    if (in_comment) {
        // We use '\"' to safely print quotation marks in the error message
        throw runtime_error("Syntax Error: Unclosed comment detected. Missing a closing '\"\"\"'.");
    }
    
    return result;
}

// THE ORCHESTRATOR

ProgramEnvironment parse_program(const string& source_code) {
    string clean_code = remove_comments(source_code);
    
    istringstream stream(clean_code);
    string line;

    string assignment_block = "";
    string eval_block = "";
    bool in_eval_section = false;
    bool first_eval_line_checked = false;

    ProgramEnvironment env;
    env.evaluation_target = nullptr;
    env.target_name = "";

    // 1. Separate the text into Assignments, Title, and Evaluation Code
    while (getline(stream, line)) {
        if (!in_eval_section && is_separator(line)) {
            in_eval_section = true;
            continue;
        }
        
        if (in_eval_section) {
            string trimmed = trim(line);
            if (trimmed.empty()) continue; 

            if (!first_eval_line_checked) {
                first_eval_line_checked = true;
                
                if (trimmed.find('\\') == string::npos && 
                    trimmed.find("λ") == string::npos && 
                    trimmed.find('(') == string::npos &&
                    trimmed.find('.') == string::npos &&
                    trimmed.find("lambda") == string::npos) {
                    
                    env.target_name = trimmed; 
                    continue; 
                }
            }
            
            eval_block += line + " ";
        } else {
            assignment_block += line + " ";
        }
    }

    // 2. Parse the Assignments (Statements ending in ';')
    size_t start = 0;
    size_t end = assignment_block.find(';');
    
    while (end != string::npos) {
        string stmt = trim(assignment_block.substr(start, end - start));
        
        if (!stmt.empty()) {
            size_t eq_pos = stmt.find('=');
            if (eq_pos == string::npos) throw runtime_error("Syntax Error: Missing '=' -> " + stmt);

            string var_name = trim(stmt.substr(0, eq_pos));
            string expr_str = trim(stmt.substr(eq_pos + 1));

            if (!is_valid_var_name(var_name)) throw runtime_error("Syntax Error: Invalid variable name '" + var_name + "'");
            if (expr_str.empty()) throw runtime_error("Syntax Error: Missing expression for '" + var_name + "'");

            // ==========================================
            // COLLISION CHECK ADDED HERE
            // ==========================================
            if (env.variables.find(var_name) != env.variables.end()) {
                throw runtime_error("Syntax Error: Naming collision. The variable '" + var_name + "' is already defined.");
            }

            lambda_expression* ast = parse_string(expr_str);
            if (!ast) throw runtime_error("Parse Error: Failed to build AST for '" + var_name + "'.");

            for (const pair<string, lambda_expression*>& pair : env.variables) {
                ast->substitute(pair.first, pair.second);
            }
            env.variables[var_name] = ast;
        }
        
        start = end + 1;
        end = assignment_block.find(';', start);
    }

    string dangling = trim(assignment_block.substr(start));
    if (!dangling.empty()) throw runtime_error("Syntax Error: Missing semicolon ';' after -> " + dangling);

    // 3. Parse the Final Evaluation Block
    eval_block = trim(eval_block);
    if (!eval_block.empty()) {
        env.evaluation_target = parse_string(eval_block);
        if (!env.evaluation_target) {
            throw runtime_error("Parse Error: Failed to build AST for the final evaluation block.");
        }
        
        for (const pair<string, lambda_expression*>& pair : env.variables) {
            env.evaluation_target->substitute(pair.first, pair.second);
        }
    }

    return env;
}


void ProgramEnvironment::combine_environments(ProgramEnvironment* pe) {
    if (pe == nullptr) return;

    for (const pair<string, lambda_expression*>& pair : pe->variables) {
        const string& var_name = pair.first;
        lambda_expression* incoming_ast = pair.second;

        // Check if the variable name already exists in OUR environment
        if (this->variables.find(var_name) != this->variables.end()) {
            throw runtime_error("Environment Merge Error: Naming collision detected. The variable '" + 
                                var_name + "' already exists.");
        }

        // If no collision, add it to our environment
        this->variables[var_name] = incoming_ast;
    }

    // Safety Step: Clear the incoming environment's variables so it doesn't 
    // accidentally double-delete the AST pointers when it gets destroyed later.
    pe->variables.clear();
}

/**
 * @brief Iterates through all stored variables in the environment and 
 * fully beta-reduces them to their Normal Form.
 * * This is a critical optimization pass. By ensuring all variables are in 
 * normal form, the compiler can use O(1) Alpha Equivalence structural 
 * checks during the resugaring (express) phase, rather than O(n^2) tree reductions.
 */
void ProgramEnvironment::reduce_variables() {
    // Iterate over every variable bound in the current environment
    for (pair<const string, lambda_expression*>& env_var : this->variables) {
        const string& var_name = env_var.first;
        lambda_expression* ast = env_var.second;

        lambda_expression* copy = lambda_expression::deepcopy(ast);

        if (ast == nullptr) continue; 

        try {
            // Attempt to pre-evaluate the AST completely.
            
            copy->reduce_total();
            delete ast;
            env_var.second = copy;
            
        } catch (const std::bad_alloc& e) {
            // SWALLOW MEMORY EXPLOSION: Catches heap exhaustion anywhere in the AST.
            cerr << "[Warning] Variable '" << var_name 
                 << "' triggered Exponential Blowup (ran out of RAM) during pre-computation. "
                 << "It has been safely stored unreduced." << endl;
                 
        } catch (const runtime_error& e) {
            // SWALLOW DIVERGENCE: Preserves lazy evaluation by allowing infinite 
            // combinators to exist in the environment as long as they aren't executed.
            cerr << "[Warning] Variable '" << var_name 
                 << "' failed to reduce during pre-computation (" << e.what() << "). "
                 << "It has been safely stored unreduced." << endl;
        }
    }
}

lambda_expression* ProgramEnvironment::express(lambda_expression* target) {
    if (target == nullptr) return nullptr;

    switch (target->type) {
        case NodeType::VARIABLE: 
            // Variables are already fully folded, no need to check or recurse
            return target;
            
        case NodeType::LAMBDA: 
        case NodeType::APPLICATION: {
            // 1. GREEDY PATTERN MATCHING (Maximal Munch)
            // Check if this entire subtree is beta-equivalent to a saved variable.
            for (const pair<string, lambda_expression*>& env_var : this->variables) {
                
                // Dereference the stored pointer to trigger your alpha_equivalence overload
                if (*target == *(env_var.second)) {
                    
                    // Memory Safety: Since we are replacing this entire complex subtree,
                    // we must delete it to free the heap memory (which triggers its 
                    // recursive destructor), and return a brand new, clean variable node.
                    delete target;
                    return make_var(env_var.first);
                }
            }
            // 2. RECURSION
            // If the whole block didn't match anything, step one level deeper.
            if (target->type == NodeType::LAMBDA) {
                target->expression_1 = express(target->expression_1);
            } else { // APPLICATION
                target->expression_1 = express(target->expression_1);
                target->expression_2 = express(target->expression_2);
            }
            return target;
        }
        default: 
            return target;
    }
}

#include <iostream>
#include <fstream>
#include <sstream>
#include <string>
#include "../include/environment.h"

using namespace std;

int main(int argc, char* argv[]) {
    if (argc < 2) {
        cerr << "Usage: " << argv[0] << " <filename.lc>" << endl;
        return 1;
    }

    string filepath = argv[1];
    ifstream file(filepath);
    
    if (!file.is_open()) {
        cerr << "Fatal Error: Could not open file '" << filepath << "'" << endl;
        return 1;
    }

    stringstream buffer;
    buffer << file.rdbuf();
    string source_code = buffer.str();
    file.close();

    try {
        ProgramEnvironment env = parse_program(source_code);

        cout << "=== Environment Loaded ===" << endl;
        for (const pair<string, lambda_expression*>& pair : env.variables) {
            cout << pair.first << " = " << pair.second->to_string() << endl;
        }
        cout << "==========================\n" << endl;

        if (env.evaluation_target != nullptr) {
            if (!env.target_name.empty()) {
                cout << "Evaluating: " << env.target_name << endl;
            } else {
                cout << "Evaluating: " << env.evaluation_target->to_string() << endl;
            }
            
            env.evaluation_target->reduce_total();
            lambda_expression* simplified_lambda_ex = lambda_expression::deepcopy(env.evaluation_target);
            env.reduce_variables();
            
            cout << "Result:     " << env.express(simplified_lambda_ex)->to_string() << endl;
            
            delete env.evaluation_target;
        } else {
            cout << "No execution block found. Variables loaded successfully." << endl;
        }


    } catch (const exception& e) {
        cerr << "\n[COMPILER ERROR]\n" << e.what() << endl;
        return 1;
    }

    return 0;
}

using namespace std;

/**
 * @brief Creates a deep, independent copy of an Abstract Syntax Tree (AST).
 * * In C++, if you assign one pointer to another, they both point to the same memory.
 * If one is deleted, the other becomes a dangling pointer. This function recursively
 * allocates new memory for every node in the tree to ensure complete independence.
 * * @param node A pointer to the root of the tree you want to copy.
 * @return A pointer to the root of the newly allocated tree.
 */
lambda_expression* lambda_expression::deepcopy(const lambda_expression* node) {
    if (node == nullptr) return nullptr;
    
    lambda_expression* copy = new lambda_expression();
    copy->type = node->type;
    copy->name = node->name;
    copy->expression_1 = deepcopy(node->expression_1);
    copy->expression_2 = deepcopy(node->expression_2);
    
    return copy;
}

/**
 * @brief Validates the structural integrity of the AST node.
 * * Ensures that the memory structure matches the rules of Lambda Calculus.
 * For example, a VARIABLE cannot have children, and an APPLICATION must have both.
 * * @return True if the node is mathematically sound, false if memory is corrupted or malformed.
 */
bool lambda_expression::is_valid() const {
    switch (type) {
        case NodeType::VARIABLE:
            return !name.empty() && expression_1 == nullptr && expression_2 == nullptr;
        case NodeType::LAMBDA:
            return !name.empty() && expression_1 != nullptr && expression_2 == nullptr && expression_1->is_valid();
        case NodeType::APPLICATION:
            return name.empty() && expression_1 != nullptr && expression_2 != nullptr && expression_1->is_valid() && expression_2->is_valid();
        default:
            return false;
    }
}

/**
 * @brief Checks if the current node is a Reducible Expression (Redex).
 * * A redex only exists when a function (LAMBDA) is applied to an argument.
 * Therefore, the current node must be an APPLICATION, and its left child must be a LAMBDA.
 * * @return True if the node can be beta-reduced.
 */
bool lambda_expression::is_redux() {
    return type == NodeType::APPLICATION && expression_1->type == NodeType::LAMBDA;
}

/**
 * @brief Public interface to rename free variables.
 * * Aliases to rename_free to keep the public API clean and intuitive.
 */
void lambda_expression::rename(const string& old_name, const string& new_name) {
    this->rename_free(old_name, new_name);
}

/**
 * @brief Recursively renames variables that are "free" (not bound by a local lambda).
 * * If it encounters a LAMBDA that binds the `old_name` (e.g., renaming 'x' inside \x. x),
 * it stops recursing, because the inner 'x' is shadowed and belongs to the inner lambda, 
 * not the outer scope.
 */
void lambda_expression::rename_free(const string& old_name, const string& new_name) {
    switch (type) {
        case NodeType::VARIABLE:
            if (this->name == old_name) this->name = new_name;
            break;
        case NodeType::APPLICATION:
            this->expression_1->rename_free(old_name, new_name);
            this->expression_2->rename_free(old_name, new_name);
            break;
        case NodeType::LAMBDA:
            if (this->name == old_name) {
                // Shadowing detected. Rename the binder and switch to rename_bound 
                // to force all instances in the body to match the new binder name.
                this->name = new_name;
                this->expression_1->rename_bound(old_name, new_name);
            } else {
                // Not shadowed, keep traversing normally.
                this->expression_1->rename_free(old_name, new_name);
            }
    }
}

/**
 * @brief Force-renames all occurrences of a variable name regardless of standard shadowing rules.
 * * This is primarily an internal helper used during alpha-conversion to aggressively 
 * update a lambda's bound variable and all its internal references at once.
 */
void lambda_expression::rename_bound(const string& old_name, const string& new_name) {
    switch (type) {
        case NodeType::VARIABLE:
            if (this->name == old_name) this->name = new_name;
            break;
        case NodeType::APPLICATION:
            this->expression_1->rename_bound(old_name, new_name);
            this->expression_2->rename_bound(old_name, new_name);
            break;
        case NodeType::LAMBDA:
            if (this->name != old_name) this->expression_1->rename_bound(old_name, new_name);
    }
}

/**
 * @brief Scans the tree to see if a specific variable name exists "freely".
 * * A variable is free if it is used without being wrapped in a lambda that binds it.
 * This is a crucial prerequisite check for capture-avoiding substitution.
 */
bool lambda_expression::occurs_free_in(const string& var) const {
    switch (type) {
        case NodeType::VARIABLE: return name == var;
        case NodeType::APPLICATION: return expression_1->occurs_free_in(var) || expression_2->occurs_free_in(var);
        case NodeType::LAMBDA:
            // If the lambda binds the variable we are looking for, it cannot occur free inside.
            if (name == var) return false;
            return expression_1->occurs_free_in(var);
        default: return false;
    }
}

/**
 * @brief Generates a mathematically unique variable name (Alpha-Conversion).
 * * If substituting `y` into `\y. x` causes a naming collision, this function appends 
 * apostrophes (e.g., `y'`) until it finds a name that does not exist anywhere in the 
 * current scope or the replacement term.
 */
string lambda_expression::generate_fresh_name(const string& base, const lambda_expression* replacement) const {
    string candidate = base;
    while (true) {
        bool fresh = true;
        if (expression_1 && expression_1->occurs_free_in(candidate)) fresh = false;
        if (replacement && replacement->occurs_free_in(candidate)) fresh = false;
        
        if (fresh) break;
        candidate += "'"; // Mutate and retry
    }
    return candidate;
}

/**
 * @brief The core substitution engine: replaces instances of `var` with the `replacement` tree.
 * * Features automatic capture avoidance. If a naming collision is detected, it pauses 
 * the substitution to alpha-convert (rename) the conflicting bound variables, ensuring 
 * mathematical meaning is preserved.
 */
void lambda_expression::substitute(const string& var, const lambda_expression* replacement) {
    switch (type) {
        case NodeType::VARIABLE:
            if (name == var) {
                // Deepcopy the replacement to prevent double-free memory corruption later.
                lambda_expression* temp = deepcopy(replacement);
                
                // Inherit the replacement's properties (in-place mutation)
                this->type = temp->type;
                this->name = temp->name;
                this->expression_1 = temp->expression_1;
                this->expression_2 = temp->expression_2;
                
                // Nullify the temp pointers so its destructor doesn't delete our new children.
                temp->expression_1 = nullptr;
                temp->expression_2 = nullptr;
                delete temp;
            }
            break;
        case NodeType::APPLICATION:
            expression_1->substitute(var, replacement);
            expression_2->substitute(var, replacement);
            break;
        case NodeType::LAMBDA:
            // Stop if the variable we are replacing is shadowed by this lambda.
            if (name == var) return; 
            
            // Capture Avoidance Check
            if (replacement->occurs_free_in(name)) {
                string fresh = this->generate_fresh_name(name, replacement);
                expression_1->rename_bound(name, fresh);
                name = fresh;
            }
            expression_1->substitute(var, replacement);
    }
}

/**
 * @brief Drives the reduction process until a Normal Form is reached.
 * * Uses a MAX_STEPS circuit breaker. If the evaluation loops infinitely 
 * (e.g., the Omega combinator), it catches the divergence and throws an error 
 * rather than hanging the system or overflowing the stack.
 */
void lambda_expression::reduce_total(int MAX_STEPS) {
    int steps = 0;
    try {
        while (this->reduce()) {
            if (steps >= MAX_STEPS) {
                throw runtime_error("DIVERGENCE_ERROR: Maximum reduction steps exceeded");
            }
            steps++;
        }
    } catch (const std::bad_alloc& e) {
        // C++ throws bad_alloc when the heap runs out of RAM.
        // We intercept it and throw it as a standard runtime error so the 
        // environment parser can safely swallow it!
        throw runtime_error("MEMORY_ERROR: The expression experienced exponential blowup and exhausted system memory.");
        
    } catch (const runtime_error& e) {
        // Standard timeout errors fall through here
        cerr << "Reduction Failed: " << e.what() << endl;
        throw; 
    }
}

/**
 * @brief Performs a single Normal Order (Lazy) beta-reduction step.
 * * Normal Order means we always attempt to evaluate the leftmost-outermost redex first.
 * This strategy guarantees we find a normal form if one exists, allowing us to safely 
 * drop divergent (infinite) terms if they are passed as unused arguments.
 * * @return True if a reduction occurred, false if no redexes remain.
 */
bool lambda_expression::reduce() {
    // Top-Down Check: Attempt root reduction first
    if (this->is_redux()) {
        lambda_expression* lambda = this->expression_1;
        lambda_expression* body = lambda->expression_1;
        lambda_expression* argument = this->expression_2;

        // Perform the mathematical substitution
        body->substitute(lambda->name, argument);

        // Splice the reduced body into the current node's location
        this->type         = body->type;
        this->name         = body->name;
        this->expression_1 = body->expression_1;
        this->expression_2 = body->expression_2;

        // Clean up consumed nodes safely
        body->expression_1 = nullptr;
        body->expression_2 = nullptr;
        delete body;                    
        lambda->expression_1 = nullptr; 
        delete lambda;                  
        delete argument;
        
        return true;
    } else {
        // If root is not a redex, recurse left-to-right
        switch (type) {
            case NodeType::VARIABLE: return false;
            case NodeType::LAMBDA: return this->expression_1->reduce();
            case NodeType::APPLICATION:
                // Short-circuiting: If left reduces, stop. Ensures lazy evaluation.
                if (this->expression_1->reduce()) return true;
                return this->expression_2->reduce();
        }
    }
    return false;
}

/**
 * @brief Destructor.
 * Recursively deletes child nodes. If children are set to nullptr, `delete nullptr` 
 * does nothing, making it safe.
 */
lambda_expression::~lambda_expression() {
    delete expression_1;
    delete expression_2;
}


// ==========================================
// ALPHA EQUIVALENCE HELPER
// ==========================================
static bool is_alpha_equiv(const lambda_expression* a, 
                           const lambda_expression* b, 
                           std::unordered_map<std::string, std::string>& env_A_to_B,
                           std::unordered_map<std::string, std::string>& env_B_to_A) {
                           
    // 1. Safety & Type Checks
    if (a == nullptr && b == nullptr) return true;
    if (a == nullptr || b == nullptr) return false;
    if (a->type != b->type) return false;

    switch (a->type) {
        case NodeType::VARIABLE: {
            // Explicitly defining the iterator types instead of using 'auto'
            std::unordered_map<std::string, std::string>::iterator it_A = env_A_to_B.find(a->name);
            std::unordered_map<std::string, std::string>::iterator it_B = env_B_to_A.find(b->name);
            
            bool a_is_bound = (it_A != env_A_to_B.end());
            bool b_is_bound = (it_B != env_B_to_A.end());

            if (a_is_bound && b_is_bound) {
                // Both are bound variables: they must map directly to each other
                return it_A->second == b->name;
            } else if (!a_is_bound && !b_is_bound) {
                // Neither are bound: they are free variables and must have the exact same name
                return a->name == b->name;
            } else {
                // One is bound and the other is free: automatic mismatch
                return false;
            }
        }

        case NodeType::APPLICATION:
            return is_alpha_equiv(a->expression_1, b->expression_1, env_A_to_B, env_B_to_A) &&
                   is_alpha_equiv(a->expression_2, b->expression_2, env_A_to_B, env_B_to_A);

        case NodeType::LAMBDA: {
            // STEP 1: SAVE PREVIOUS STATE (Handles variable shadowing)
            bool a_had_prev = env_A_to_B.count(a->name);
            std::string prev_A = a_had_prev ? env_A_to_B[a->name] : "";
            
            bool b_had_prev = env_B_to_A.count(b->name);
            std::string prev_B = b_had_prev ? env_B_to_A[b->name] : "";

            // STEP 2: APPLY NEW BINDINGS FOR THIS SCOPE
            env_A_to_B[a->name] = b->name;
            env_B_to_A[b->name] = a->name;
            
            // STEP 3: RECURSE
            bool bodies_match = is_alpha_equiv(a->expression_1, b->expression_1, env_A_to_B, env_B_to_A);
            
            // STEP 4: RESTORE PREVIOUS STATE (Pops the scope safely)
            if (a_had_prev) env_A_to_B[a->name] = prev_A;
            else env_A_to_B.erase(a->name);
            
            if (b_had_prev) env_B_to_A[b->name] = prev_B;
            else env_B_to_A.erase(b->name);
            
            return bodies_match;
        }
    }
    return false;
}

// ==========================================
// EQUALITY OPERATOR OVERLOAD
// ==========================================
bool lambda_expression::operator==(const lambda_expression& other) const {
    // Instantiate the dual maps for O(1) bijection lookups
    std::unordered_map<std::string, std::string> env_A_to_B;
    std::unordered_map<std::string, std::string> env_B_to_A;
    
    // Fire the recursive helper
    return is_alpha_equiv(this, &other, env_A_to_B, env_B_to_A);
}

/**
 * @brief Serializes the AST back into a human-readable mathematical string.
 */
string lambda_expression::to_string() const {
    switch (type) {
        case NodeType::VARIABLE: return name;
        case NodeType::LAMBDA: return "(λ" + name + ". " + expression_1->to_string() + ")";
        case NodeType::APPLICATION: return "(" + expression_1->to_string() + " " + expression_2->to_string() + ")";
        default: return "<malformed>";
    }
}
bool lambda_expression::beta_equivalence(const lambda_expression& other, int max_depth) const {
    // 1. Pass the pointers to your static deepcopy function
    lambda_expression* le_0 = deepcopy(this);
    lambda_expression* le_1 = deepcopy(&other);

    bool is_equiv = false;

    // 2. Wrap the reduction in a try-catch to ensure memory safety
    try {
        // Pass the max_depth safeguard down the chain
        le_0->reduce_total(max_depth);
        le_1->reduce_total(max_depth);

        // 3. Dereference the pointers to trigger your overloaded Alpha equivalence operator
        is_equiv = (*le_0 == *le_1);
        
    } catch (...) {
        // If reduce_total hits an infinite loop and throws an error,
        // we MUST clean up the copied trees before passing the error up!
        delete le_0;
        delete le_1;
        throw; // Re-throw the original error to the user
    }

    // 4. Standard memory cleanup for a successful run
    delete le_0;
    delete le_1;

    return is_equiv;
}

/** @brief Instantiates a basic variable node on the heap. */
lambda_expression* make_var(const string& name) {
    lambda_expression* node = new lambda_expression();
    node->type = NodeType::VARIABLE;
    node->name = name;
    node->expression_1 = nullptr;
    node->expression_2 = nullptr;
    return node;
}

/** @brief Instantiates a lambda abstraction, capturing the given body tree. */
lambda_expression* make_lambda(const string& bound_var, lambda_expression* body) {
    lambda_expression* node = new lambda_expression();
    node->type = NodeType::LAMBDA;
    node->name = bound_var;
    node->expression_1 = body;
    node->expression_2 = nullptr;
    return node;
}

/** @brief Instantiates an application of a function to an argument. */
lambda_expression* make_app(lambda_expression* func, lambda_expression* arg) {
    lambda_expression* node = new lambda_expression();
    node->type = NodeType::APPLICATION;
    node->name = "";
    node->expression_1 = func;
    node->expression_2 = arg;
    return node;
}

#include "../include/lambda_parser.h"
#include <unordered_set>
#include <stdexcept>
#include <iostream>

using namespace std;

// ==========================================
// STEP 1: THE LEXER (Tokenization)
// ==========================================

/**
 * @brief Converts a continuous string of text into discrete, actionable tokens.
 * * The Lexer scans byte-by-byte. It ignores whitespace, treats special characters 
 * as standalone delimiters, and continuously accumulates alphanumeric characters 
 * into complete variable names (e.g., separating "var_x" from ".").
 * * @param str The raw input string from the user or file.
 * @return A vector of clean strings representing syntax tokens.
 */
vector<string> tokenize(const string& str) {
    vector<string> tokens;
    string curr_word = "";
    
    // The exact characters that trigger a split in Lambda Calculus grammar
    unordered_set<char> delimiters = {' ', '.', '(', ')', ';', '\\'};

    for (char c : str) {
        if (delimiters.count(c)) {
            // If we were building a variable name, save it before processing the delimiter
            if (!curr_word.empty()) {
                tokens.push_back(curr_word);
                curr_word = "";
            }
            // Discard spaces, but keep mathematical symbols
            if (c != ' ') tokens.push_back(string(1, c));
        } else {
            // Not a delimiter; accumulate into the current variable name
            curr_word += c;
        }
    }
    
    // Flush any remaining characters in the buffer
    if (!curr_word.empty()) tokens.push_back(curr_word);
    
    return tokens;
}

// ==========================================
// STEP 2: THE PARSER (Recursive Descent)
// ==========================================

/**
 * @brief The engine that translates a linear list of tokens into a hierarchical AST.
 * * This class uses Recursive Descent. It consumes tokens from left to right and 
 * relies on the C++ call stack (recursion) to handle nested parenthesis and scope, 
 * eliminating the need to manually manage data stacks and booleans.
 */
class LambdaParser {
private:
    vector<string> tokens;
    size_t pos;

    /** @brief Looks at the current token without consuming it. */
    string peek() { return (pos >= tokens.size()) ? "" : tokens[pos]; }
    
    /** @brief Consumes the current token and moves the internal cursor forward. */
    string advance() { return (pos >= tokens.size()) ? "" : tokens[pos++]; }

    /**
     * @brief Parses a single self-contained unit (Atom).
     * * An atom is either a variable, an entire parenthesized expression, or a lambda.
     * This function contains the logic for Greedy Lambda Scope and Automatic Currying.
     */
    lambda_expression* parse_atom() {
        string t = peek();

        // Rule 1: Parentheses override scope. Recurse entirely until the matching ')'
        if (t == "(") {
            advance();
            lambda_expression* expr = parse_expression();
            if (advance() != ")") throw runtime_error("Parse Error: Missing closing ')'");
            return expr;
        } 
        
        // Rule 2: Lambda functions
        else if (t == "lambda" || t == "λ" || t == "\\") {
            advance();
            
            // Automatic Currying: Collect all variables (e.g., \x y z .)
            vector<string> bound_vars;
            while (peek() != "." && peek() != "") bound_vars.push_back(advance());
            
            if (bound_vars.empty()) throw runtime_error("Parse Error: Lambda missing variable");
            if (advance() != ".") throw runtime_error("Parse Error: Expected '.' after variables");

            // Greedy Scope: The body consumes everything else remaining in the expression
            lambda_expression* body = parse_expression();
            
            // Unpack the curried variables inside-out to build the nested tree
            for (int i = bound_vars.size() - 1; i >= 0; --i) {
                body = make_lambda(bound_vars[i], body);
            }
            return body;
        } 
        
        // Rule 3: End of statement marker
        else if (t == ";") {
            advance();
            return nullptr;
        }
        
        // Rule 4: If it's none of the above, it must be a standard variable
        else {
            advance();
            return make_var(t);
        }
    }

public:
    LambdaParser(const vector<string>& t) : tokens(t), pos(0) {}

    /**
     * @brief Parses a full expression, enforcing Left-Associativity.
     * * This function continuously applies new arguments to the existing expression
     * from left to right. E.g., reading "x y z" results in ((x y) z).
     */
    lambda_expression* parse_expression() {
        if (peek() == "") return nullptr;
        
        // Obtain the base function
        lambda_expression* expr = parse_atom();

        // Continuously consume remaining atoms and apply them to the base
        while (peek() != "" && peek() != ")" && peek() != ";") {
            lambda_expression* next_arg = parse_atom();
            if (next_arg) expr = make_app(expr, next_arg);
        }
        
        return expr;
    }
};

// ==========================================
// STEP 3: THE WRAPPER (Public API)
// ==========================================

/**
 * @brief Safely orchestrates the Lexing and Parsing processes.
 * * Provides a unified entry point for the rest of the program to convert strings 
 * into ASTs. Wraps the engine in a try-catch block to prevent syntax errors 
 * from causing fatal C++ segmentation faults.
 */
lambda_expression* parse_string(const string& str) {
    try {
        // Pass 1: Lexing
        vector<string> tokens = tokenize(str);
        if (tokens.empty()) return nullptr;

        // Pass 2: Parsing
        LambdaParser parser(tokens);
        return parser.parse_expression();
        
    } catch (const exception& e) {
        // Log syntax errors gracefully
        cout << e.what() << endl;
        return nullptr;
    }
}

    void report(bool pass, const string& name, const string& input,

        if (pass) {
            ++tests_passed;
            if (verbosity != Verbosity::QUIET) cout << "PASS " << name << "\n";

            ++tests_failed;
            if (verbosity != Verbosity::QUIET) cout << "FAIL " << name << "\n";

            cout << "   Input:     " << input << "\n";
            cout << "   Operation: " << op << "\n";
            cout << "   Expected:  " << expected << "\n";
            cout << "   Actual:    " << actual << "\n";

        }

        streambuf* original_cout = cout.rdbuf();

        // 2. If in quiet mode, redirect all 'cout' to the black hole

            cout.rdbuf(dummy_buffer.rdbuf());
        }

            cout.rdbuf(original_cout);

            report(expected == "ERROR", test_name, input, "parse_string()", expected, "nullptr (Caught Error)");
        } else {
            string actual = expr->to_string();
            report(actual == expected, test_name, input, "parse_string()", expected, actual);

        }

    for (const TestModule& module : test_suite) {
        if (!quiet) cout << "\n>>> Running " << module.name << " Tests...\n";
        
        // Execute the test and unpack the pair
        pair<int, int> results = module.run(verbose, quiet);
        total_passed += results.first;
        total_failures += results.second;
    }

    // ANSI Color Codes
    const string RESET = "\033[0m";
    const string BOLD_GREEN = "\033[1;32m";
    const string BOLD_RED = "\033[1;31m";

    if (!quiet) cout << "\n#############################################\n";
    
    // 4. Print the final aggregated summary in color
    if (total_failures == 0) {
        cout << BOLD_GREEN << "   ALL MODULES PASSED! SYSTEM IS STABLE.   \n";
        cout << "   (Total: " << total_passed << " tests passed)\n" << RESET;
    } else {
        cout << BOLD_RED << "   SYSTEM FAILED WITH " << total_failures << " ERRORS.\n";
        cout << "   (" << total_passed << " tests passed)\n" << RESET;
    }
    
    if (!quiet) cout << "#############################################\n";

    return total_failures > 0 ? 1 : 0;
}
